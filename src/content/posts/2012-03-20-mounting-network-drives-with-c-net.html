First off, big thanks to aejw’s NetworkDrive class for making my life easier. Here’s how I use it to map a share from C#. Call the helper with a drive letter and a share path. If you leave out the username, it lets NetworkDrive decide which credentials to use. You can also pass a username and password. If the code throws while setting up or mapping the drive, you get false back.

Open the original NetworkDrives0015 source ZIP, then extract cNetworkDrives0015.cs and add it to your project.

Add the drive-mapping helper

using aejw.Network;


private bool mount_drive(string localdrive, string sharelocation, string user = "blank", string pass = "blank")
        {

            NetworkDrive oNetDrive = new NetworkDrive();

            try
            {
                oNetDrive.LocalDrive = localdrive;
                oNetDrive.ShareName = sharelocation;
                if (user == "blank")
                {
                    oNetDrive.MapDrive();
                }
                else
                {
                    oNetDrive.MapDrive(user, pass);
                }
            }
            catch (Exception err)
            {
                //Optional MessageBox to show Error
                //MessageBox.Show(this, "Error: " + err.Message);
                return false;
            }

            return true;

        }
